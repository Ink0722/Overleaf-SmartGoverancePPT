\section{Research Design}
\begin{frame}[t]{A 2 $\times$ 2 Field Experiment}
\small

\begin{block}{Assignment and sample}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}lXX@{}}\toprule
&\textbf{Without AI}&\textbf{With AI}\\\midrule
Individual&154 people&155 people\\
Two-person team&115 pairs / 230 people&126 pairs / 252 people\\\bottomrule
\end{tabularx}
\end{block}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Team composition}
Each pair combines one R\&D and one commercial employee. Participants address business challenges within their own units.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Analytical sample}
791 participants produce 550 submissions. Each pair submits one joint solution for expert assessment.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

\begin{frame}[t]{Experimental Workflow}
\small
\begin{block}{Prepare and assign}
Participants receive task guidance and are randomly assigned to a condition. AI groups receive prompting training and a GPT-4-based tool.
\end{block}
\begin{block}{Generate, select and develop}
Every individual or pair generates five ideas, selects one and develops it into a final proposal. This sequence separates idea generation from selection.
\end{block}
\begin{alertblock}{Evaluate and measure}
Independent experts assess quality and technicality. Surveys record emotional change, while interaction records reveal patterns of AI use.
\end{alertblock}

\end{frame}

\begin{frame}[t]{Measures and Estimation}
\small
\begin{columns}[T,onlytextwidth]
\begin{column}{.52\textwidth}
\begin{block}{Outcome measures}
\begin{itemize}
\item \textbf{Quality} uses expert ratings on a 1--10 scale, standardized against the control group.
\item \textbf{Technicality} ranges from commercial (1) to technical (7).
\item \textbf{Emotions} measure changes between pretask and post-task surveys.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.44\textwidth}
\begin{alertblock}{Treatment comparisons}
Regression coefficients compare each condition with individuals working without AI. Additional specifications include business-unit and date effects, employee characteristics and prior AI use.
\end{alertblock}
\end{column}
\end{columns}
