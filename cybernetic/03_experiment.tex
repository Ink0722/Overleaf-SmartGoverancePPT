\begin{frame}\frametitle{Field Setting: P\&G Product Innovation}
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}\begin{block}{Setting}\begin{itemize}\item 791 randomized professionals.\item R\&D and commercial functions.\item Four business units in Europe and the Americas.\item May--July 2024.\end{itemize}\end{block}\end{column}
\begin{column}{.49\textwidth}\begin{alertblock}{Task}\begin{itemize}\item Address real product-development needs.\item Combine technical and market considerations.\item Submit proposals for independent expert assessment.\end{itemize}\end{alertblock}\end{column}

\begin{frame}\frametitle{Estimating Treatment Effects}
\begin{block}{Regression specification}
\scriptsize\[
Y_i=\beta_0+\beta_1\mathrm{TeamNoAI}_i+\beta_2\mathrm{AloneAI}_i+
\beta_3\mathrm{TeamAI}_i+\gamma X_i+\delta FE_i+\varepsilon_i.
\]
\end{block}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}\begin{itemize}\item \textbf{Baseline:} individual without AI.\item Each coefficient compares a condition with the baseline.\end{itemize}\end{column}
\begin{column}{.48\textwidth}\begin{itemize}\item Fixed effects for business unit and date.\item Controls: experience, job level, gender, previous AI use.\end{itemize}\end{column}
\end{columns}\src{Section 4.1}\end{frame}

\begin{frame}\frametitle{Why the Comparison Is Informative}
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}\begin{block}{Design safeguards}\begin{itemize}\item Stratified random assignment.\item Evaluators blind to experimental group.\item Covariate adjustment and robustness checks.\end{itemize}\end{block}\end{column}
\begin{column}{.49\textwidth}\begin{alertblock}{Do not confuse the units}\begin{itemize}\item \textbf{791 people} randomized.\item \textbf{550 solutions} in quality regressions.\item Each two-person team submits one solution.\end{itemize}\end{alertblock}\end{column}
\end{columns}\src{Sections 3--4; Tables 1--2}\end{frame}
