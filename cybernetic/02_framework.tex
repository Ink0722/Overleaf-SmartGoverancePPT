\section{Conceptual Framework}
\begin{frame}[t]{Three Dimensions of Collaboration}
\small
\vspace{0.22cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.31\textwidth}
{\Large\textcolor{paperred}{\bfseries 01}}\par\medskip
{\bfseries Performance}\par\smallskip
Does AI improve expert-rated solution quality as much as a human teammate?
\end{column}
\begin{column}{.31\textwidth}
{\Large\textcolor{paperred}{\bfseries 02}}\par\medskip
{\bfseries Expertise}\par\smallskip
Can AI help employees cross the boundary between technical and commercial knowledge?
\end{column}
\begin{column}{.31\textwidth}
{\Large\textcolor{paperred}{\bfseries 03}}\par\medskip
{\bfseries Sociality}\par\smallskip
Does working with AI change enthusiasm, energy, anxiety or frustration during the task?
\end{column}
\end{columns}

\vspace{0.85cm}
\begin{block}{What the comparison reveals}
The same four experimental conditions test whether AI changes the value of working alone and the value of working with another person.
\end{block}
\end{frame}
