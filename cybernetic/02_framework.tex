\section{Conceptual Framework}
\begin{frame}[t]{Three Dimensions of Collaboration}
\small
\begin{block}{Performance}
Human teamwork pools perspectives to improve solutions. The study tests whether AI assistance can provide comparable gains in expert-rated quality.
\end{block}
\begin{block}{Expertise}
Cross-functional collaboration combines technical and commercial knowledge. The study examines whether AI broadens the orientation of employees' proposals.
\end{block}
\begin{alertblock}{Sociality}
Working with others can support motivation. Changes in enthusiasm, energy, anxiety and frustration capture the emotional experience of working with AI.
\end{alertblock}

\end{frame}
