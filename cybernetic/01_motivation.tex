\section{Research Overview}
\begin{frame}[t]{The Cybernetic Teammate}
\small
\begin{block}{Research question}
Can generative AI provide the performance, expertise and social benefits associated with human teamwork?
\end{block}

\vspace{0.28cm}
\centering
\textcolor{paperred}{\bfseries A field experiment with 791 P\&G professionals}\par\smallskip
\renewcommand{\arraystretch}{1.45}
\begin{tabularx}{0.84\linewidth}{@{}>{\bfseries}lXX@{}}
& \centering\arraybackslash Without AI & \centering\arraybackslash With AI \\
\midrule
Individual & \centering\arraybackslash Solo work & \centering\arraybackslash AI-assisted individual \\
Two-person team & \centering\arraybackslash Human teamwork & \centering\arraybackslash Human + AI teamwork \\
\bottomrule
\end{tabularx}

\vspace{0.34cm}
\begin{alertblock}{Central result}
AI-assisted individuals reach solution quality comparable to human-only teams, while gaining broader perspectives and a more positive task experience.
\end{alertblock}
\end{frame}

\begin{frame}[t]{Motivation and Research Contribution}
\small
\textcolor{paperred}{\bfseries Product innovation needs more than one kind of expertise.}\par\medskip
Technical feasibility and market relevance often sit in different functions. Cross-functional teams bring them together, but require coordination between people.

\vspace{0.48cm}
\textcolor{paperred}{\bfseries Generative AI may change this division of work.}\par\medskip
An individual can consult AI for ideas outside their specialization. The experiment asks which benefits of collaboration this access can reproduce or extend.

\vspace{0.50cm}
\begin{alertblock}{Contribution}
By varying both human teamwork and AI access, the study connects solution quality with expertise integration and emotional experience.
\end{alertblock}
\end{frame}
