\section{Research Overview}
\begin{frame}[t]{The Cybernetic Teammate}
\small
\begin{block}{Research question}
Can generative AI provide the performance, expertise and social benefits associated with human teamwork?
\end{block}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Field experiment}
At P\&G, 791 professionals worked on real innovation challenges, individually or in cross-functional pairs, with or without AI.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Main finding}
AI-assisted individuals achieved quality comparable to human-only teams, while producing broader solutions and reporting a more positive work experience.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

\begin{frame}[t]{Motivation and Research Contribution}
\small
\begin{block}{Why organizations form teams}
Product innovation combines technical feasibility with market relevance. Human teams bring these perspectives together, but coordination and functional boundaries can constrain collaboration.
\end{block}
\begin{block}{What AI changes}
Generative AI offers employees access to ideas and knowledge beyond their usual specialization. This raises a question about how AI changes the benefits of working with another person.
\end{block}
\begin{alertblock}{Contribution}
The experiment varies both human teamwork and AI access, linking output quality to expertise integration and emotional experience.
\end{alertblock}

\end{frame}
